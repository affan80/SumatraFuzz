cmake_minimum_required(VERSION 3.20)
project(sumatrafuzz_worker_supervisor LANGUAGES CXX)

add_executable(command_line_tests src/command_line.cpp tests/command_line_tests.cpp)
target_include_directories(command_line_tests PRIVATE include)
target_compile_features(command_line_tests PRIVATE cxx_std_17)
add_executable(service_host_contract_tests src/service_host.cpp tests/service_host_contract_tests.cpp)
target_include_directories(service_host_contract_tests PRIVATE include)
target_compile_features(service_host_contract_tests PRIVATE cxx_std_17)
add_executable(process_probe tests/process_probe.cpp src/command_line.cpp)
target_include_directories(process_probe PRIVATE include)
target_compile_features(process_probe PRIVATE cxx_std_17)
if(WIN32)
  add_executable(sumatrafuzz_worker_service_host src/service_host.cpp src/command_line.cpp)
  target_include_directories(sumatrafuzz_worker_service_host PRIVATE include)
  target_compile_features(sumatrafuzz_worker_service_host PRIVATE cxx_std_17)
  target_compile_definitions(sumatrafuzz_worker_service_host PRIVATE SUMATRAFUZZ_SERVICE_HOST_STANDALONE)
  target_link_libraries(sumatrafuzz_worker_service_host PRIVATE advapi32)
  set_target_properties(sumatrafuzz_worker_service_host PROPERTIES OUTPUT_NAME "SumatraFuzzWorkerHost")
endif()
if(MSVC)
  foreach(target command_line_tests service_host_contract_tests)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  endforeach()
  if(WIN32)
    target_compile_options(sumatrafuzz_worker_service_host PRIVATE /W4 /WX /permissive-)
  endif()
endif()
enable_testing()
add_test(NAME command_line_tests COMMAND command_line_tests)
add_test(NAME service_host_contract_tests COMMAND service_host_contract_tests)
